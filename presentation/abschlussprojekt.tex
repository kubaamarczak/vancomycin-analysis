    \begin{itemize}
    \item IDEEN
        \item Wie effektiv ist die Behandlung mit Vancomycin? [Vancomycin-Therapie, Nephrotoxine (nierenschädigende Arzneien)]
        \item[$\rightarrow$] mit Kormobiditäten [Komorbiditäten (Begleiterkrankungen)]
        \item[$\rightarrow$] ohne Kormobiditäten
        \item Inwiefern spielt die Dosis von Vancomycin eine Rolle?
        \item[$\rightarrow$] Unterdosis -> Resistenz [Mikrobiologische Befunde]
        \item[$\rightarrow$] Überdosis -> Nierenschäden [Nierenfunktion]
        \item Inwiefern werden die Nieren beansprucht bei Zugabe von anderen Arzneien? [Nephrotoxine (nierenschädigende Arzneien)]
        \item Inwiefern hängt die Schwere der Erkrankung von einer erfolgreichen Behandlung mit Vancomycin ab? [Schweregrad der Erkrankung]
        \item Für welche Beschwerden funktioniert Vancomycin am besten/bzw. weniger gut? [Indikationen]
        
        
    \end{itemize}
\end{frame}
